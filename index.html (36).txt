<script>
(function() {
    document.addEventListener("DOMContentLoaded", function() {
        var fileIn = document.getElementById('audio-input');
        var upText = document.getElementById('upload-text');
        var pZone = document.getElementById('process-zone');
        var origAudio = document.getElementById('audio-player');
        var cleanBtn = document.getElementById('clean-btn');
        var statusMsg = document.getElementById('loading-status');
        var dlZone = document.getElementById('download-zone');
        var resPlayer = document.getElementById('result-player');
        var dlAction = document.getElementById('download-action');
        var loadedBuffer = null;

        if(!fileIn) return;

        fileIn.addEventListener('change', function(e) {
            var file = e.target.files[0];
            if(file) {
                upText.innerText = "تم اختيار: " + file.name;
                origAudio.src = URL.createObjectURL(file);
                pZone.style.display = 'block';
                dlZone.style.display = 'none';
                statusMsg.innerText = "جاري تجهيز الملف فوريّاً...";
                
                var reader = new FileReader();
                reader.onload = function(evt) {
                    var actx = new (window.AudioContext || window.webkitAudioContext)();
                    actx.decodeAudioData(evt.target.result, function(buf) {
                        loadedBuffer = buf;
                        statusMsg.innerText = "الملف جاهز الآن للفلترة والتنقية.";
                    }, function() {
                        statusMsg.innerText = "عذراً، يرجى تجربة ملف MP3 أو WAV آخر.";
                    });
                };
                reader.readAsArrayBuffer(file);
            }
        });

        cleanBtn.addEventListener('click', function() {
            if(!loadedBuffer) return;
            statusMsg.innerText = "جاري تصفية ترددات الصدى والارتداد...";
            
            var octx = new OfflineAudioContext(loadedBuffer.numberOfChannels, loadedBuffer.length, loadedBuffer.sampleRate);
            var src = octx.createBufferSource();
            src.buffer = loadedBuffer;
            
            var hpf = octx.createBiquadFilter();
            hpf.type = "highpass"; hpf.frequency.value = 135;
            
            var peak = octx.createBiquadFilter();
            peak.type = "peaking"; peak.frequency.value = 360; peak.Q.value = 1.3; peak.gain.value = -9;
            
            var comp = octx.createDynamicsCompressor();
            comp.threshold.value = -22; comp.knee.value = 22; comp.ratio.value = 7; comp.attack.value = 0.005; comp.release.value = 0.08;
            
            src.connect(hpf); hpf.connect(peak); peak.connect(comp); comp.connect(octx.destination);
            src.start(0);
            
            octx.startRendering().then(function(rendered) {
                var numOfChan = rendered.numberOfChannels,
                    len = rendered.length * numOfChan * 2 + 44,
                    ab = new ArrayBuffer(len), view = new DataView(ab),
                    ch = [], i, sample, offset = 0, pos = 0;
                
                view.setUint32(0, 0x46464952, true); view.setUint32(4, len - 8, true); view.setUint32(8, 0x45564157, true);
                view.setUint32(12, 0x20746d66, true); view.setUint32(16, 16, true); view.setUint16(20, 1, true);
                view.setUint16(22, numOfChan, true); view.setUint32(24, rendered.sampleRate, true);
                view.setUint32(28, rendered.sampleRate * 2 * numOfChan, true); view.setUint16(32, numOfChan * 2, true);
                view.setUint16(34, 16, true); view.setUint32(36, 0x61746164, true); view.setUint32(40, len - 44, true);
                
                for (i = 0; i < numOfChan; i++) ch.push(rendered.getChannelData(i));
                pos = 44;
                while (pos < len && offset < rendered.length) {
                    for (i = 0; i < numOfChan; i++) {
                        sample = Math.max(-1, Math.min(1, ch[i][offset]));
                        sample = sample < 0 ? sample * 0x8000 : sample * 0x7FFF;
                        view.setInt16(pos, sample, true); pos += 2;
                    }
                    offset++;
                }
                var blob = new Blob([ab], {type: "audio/wav"});
                var url = URL.createObjectURL(blob);
                resPlayer.src = url; dlAction.href = url;
                dlZone.style.display = 'block';
                statusMsg.innerText = "تمت التنقية بنجاح!";
            });
        });
    });
})();
</script>
